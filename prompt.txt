# Valora cualitativaMENTE de las directrices empleadas (contesta por cada una que venga al caso si se ha aplicado o no) por el agente:
valoracion_qualitativa:
  directrices_generales:
    - Solo muestra proactividad para presupuestar en neumáticos


  # Si tipologia es "quiere_ir_al_taller" && quiere_ir_al_taller = "Nueva cita" && tipologia_reparacion contiene neumáticos:
  directrizes_neumaticos:
    - Habla de una marca de las 3 siguientes (RoadX, Falken, Pirelli)
    - Menciona que el precio incluye montaje, equilibrado, valvula y IVA
    - Menciona que el precio no incluye alineación
    - Da al taller amplitud para el tiempo de reparación

  # Si es un pinchazo:
  directrizes_pinchazo:
    - Averigua si la rueda pierde aire o el vehiculo no puede circular
    - Da al taller amplitud para el tiempo de reparación
    - Si el cliente insiste

  # Si tipologia es "quiere_ir_al_taller" && quiere_ir_al_taller = "Nueva cita" && tipologia_reparacion contiene mecanica ligera:
  directrizes_mecanica_ligera:
    - Identifica correctamente el vehiculo
    - Concreta cita sin dar precio
    - Da al taller amplitud para el tiempo de reparación

  # Si tipologia es "quiere_ir_al_taller" && quiere_ir_al_taller = "Nueva cita" && tipologia_reparacion contiene mecanica pesada:
  directrizes_mecanica_pesada:
    - No diagnostica por telefono, pasa a taller
    - No se da precio ni tiempo de reparación
    - Da a entender al cliente que el vehiculo debe quedar un tiempo prudencial en el taller

  # Si el vehiculo es de renting:
    - Da a entender al cliente que se debe pasar a taller para solicitar autorizacion al renting
    - Da a entender que pasa para solicitar autorizacion y no para la reparación

  # Si tipologia es coche_en_taller && consulta_estado:
    - Existe satisfacción del cliente en la gestion
    - Da a entender al cliente que si no se cumplira el terminio se le avisarà.
